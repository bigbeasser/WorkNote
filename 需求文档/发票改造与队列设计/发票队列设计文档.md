# 发票队列设计文档

## 数据库表设计

```sql
create table ic_invoice_generate_task
(
    id            bigint auto_increment
        primary key,
    business_id   bigint  not null comment '业务ID,入库登记ID,等',
    business_numbers varchar(100) DEFAULT NULL COMMENT '业务单号 (冗余字段，方便查询)',
    request_params   text null comment '原始请求参数 (JSON)',
    action_type  varchar(100) not null comment '发票类型,冲销发票/正向发票',
    invoice_type  varchar(100) not null comment '发票种类,采购发票/销售发票/Credit/Debit',
    source_type      varchar(500) not null comment '任务来源,crm_接收点价单/crm_接收入库登记',
    status      varchar(50) not null comment 'PENDING：待处理;PROCESSING：处理中;SUCCESS：成功;FAILED：失败;',
    sap_push_status      varchar(50)  null comment 'SUCCESS：成功;FAILED：失败;init:未推送',
    error_message   text  null comment '最近一次失败原因摘要',
    start_time datetime DEFAULT NULL COMMENT '开始执行时间',
    end_time datetime DEFAULT NULL COMMENT '结束时间',
    version int(11) NOT NULL DEFAULT 0 COMMENT '版本号 (乐观锁)',
    created_by    varchar(255) not null comment '创建者',
    created_time  datetime     not null comment '创建时间',
    inactive_flag bit          not null comment '逻辑删除标识(0:未删除、1:已删除)',
    KEY idx_business_id (business_id),
    KEY idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 comment 'IC发票自动生成队列表';


create table ic_invoice_generate_task_result
(
    id            bigint auto_increment
        primary key,
    ic_invoice_generate_task_id  bigint not null comment 'IC发票自动生成队列表id',
    business_id    bigint not null comment '业务ID,例如入库登记ID',
    relate_invoice_id     bigint not null comment '关联生成的发票',
    relate_invoice_document_id     bigint not null comment '关联生成的发票明细',
    created_by    varchar(255) not null comment '创建者',
    created_time  datetime     not null comment '创建时间',
    KEY idx_ic_invoice_generate_task_id (ic_invoice_generate_task_id),
    KEY idx_business_id (business_id),
    KEY idx_invoice_document_id (relate_invoice_document_id),
    KEY idx_invoice_id (relate_invoice_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 comment 'IC发票自动生成队列关联结果表';
```
